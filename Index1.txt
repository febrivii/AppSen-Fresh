        function getFormattedDate(dateObj) {
            const d = new Date(dateObj);
            const day = String(d.getDate()).padStart(2, '0');
            const month = String(d.getMonth() + 1).padStart(2, '0');
            const year = d.getFullYear();
            return `${day}-${month}-${year}`;
        }

        async function fetchMasterData() {
            const overlay = document.getElementById('login-overlay');
            try {
                let res = await fetch(SCRIPT_URL + "?action=getMasterData");
                let textResult = await res.text();
                
                let json;
                try {
                    json = JSON.parse(textResult);
                } catch(parseErr) {
                    throw new Error("Server mengembalikan HTML/Error bukan JSON.");
                }

                if(json.status === "success") {
                    masterStores = json.data.stores || [];
                    masterPromotors = json.data.promotors || [];
                    masterTLs = json.data.tls || [];
                    populateStoreDropdowns();
                    
                    if (overlay) overlay.classList.add('hidden');
                } else {
                    throw new Error(json.message || "Gagal memuat data master");
                }
            } catch(e) {
                console.error("Gagal memuat master data:", e);
                if (overlay) {
                    overlay.innerHTML = `
                        <div class="w-10 h-10 rounded-2xl bg-red-50 text-red-600 flex items-center justify-center mb-2 mx-auto">

        function handleLogin(event) {
            event.preventDefault();
            const nrkInput = document.getElementById('login-nrk').value.trim();
            
            let role = null;
            let name = null;

            const promotorFound = masterPromotors.find(p => String(p.nrk).trim() === nrkInput);
            if (promotorFound) {
                role = 'promotor';
                name = promotorFound.name;
            } else {
                const tlFound = masterTLs.find(t => String(t.nrk).trim() === nrkInput);
                if (tlFound) {
                    role = 'tl';
                    name = tlFound.name;
                    currentTLName = tlFound.name;
                }
            }

            if (!role) {
                showModal('Gagal Masuk', 'NRK "' + nrkInput + '" tidak ditemukan di database Master STO AOP.');
                return;
            }

            currentUser = name;
            currentRole = role;

            document.getElementById('current-user-display').innerText = name;
            document.getElementById('user-badge').classList.remove('hidden');
            document.getElementById('login-screen').classList.add('hidden');

            if(role === 'promotor') {
                document.getElementById('promotor-dashboard').classList.remove('hidden');
                document.getElementById('welcome-promotor').innerText = 'Halo, ' + name;
            } else {
                document.getElementById('tl-dashboard').classList.remove('hidden');
                document.getElementById('welcome-tl').innerText = 'Halo, ' + name;
                document.getElementById('pjp-tl-name').value = name;
                
                populateTLStoreDropdown(name);

                const myStores = masterStores.filter(store => String(store.tlName).trim().toLowerCase() === String(name).trim().toLowerCase());
                const regionVal = myStores.length > 0 ? (myStores[0].region || 'Nasional') : 'Area Mandiri';
                document.getElementById('pjp-region').value = regionVal;
                
                initPJPGeoLocation();
            }
        }

        function switchTLTab(menu) {
            const menus = ['pjptl', 'mypjp', 'pantau', 'performance'];
            menus.forEach(m => {
                const tabEl = document.getElementById('tl-tab-' + m);
                const btnEl = document.getElementById('tl-btn-' + m);
                if(m === menu) {
                    tabEl.classList.remove('hidden');
                    btnEl.className = 'tl-menu-btn p-3 bg-blue-600 text-white rounded-2xl text-xs font-bold shadow-lg shadow-blue-600/25 transition flex flex-col items-center justify-center space-y-1 border border-blue-500 active';
                    const iconEl = btnEl.querySelector('i');
                    if(iconEl) iconEl.className = iconEl.className.replace(/text-blue-600/g, 'text-blue-200');
                } else {
                    tabEl.classList.add('hidden');
                    btnEl.className = 'tl-menu-btn p-3 bg-white text-slate-700 hover:bg-slate-50 rounded-2xl text-xs font-bold shadow-sm border border-slate-200 transition flex flex-col items-center justify-center space-y-1';
                    const iconEl = btnEl.querySelector('i');
                    if(iconEl) iconEl.className = iconEl.className.replace(/text-blue-200/g, 'text-blue-600');
                }
            });

            if(menu === 'mypjp') loadMyPJP();
            if(menu === 'pantau') loadPantauData();
            if(menu === 'performance') loadPerformanceData();
        }

        async function loadMyPJP() {
            const container = document.getElementById('mypjp-content');
            container.innerHTML = '<p class="text-slate-500 text-center py-6 font-medium">Memuat data PJP...</p>';
            try {
                let res = await fetch(SCRIPT_URL + "?action=getPJPData");
                let textRes = await res.text();
                let json = JSON.parse(textRes);
                if(json.status === "success" && json.data.length > 0) {
                    myPJPCachedRecords = json.data.filter(row => String(row.tlName).trim().toLowerCase() === String(currentUser).trim().toLowerCase());
                    
                    const validPJPCount = myPJPCachedRecords.filter(row => {
                        const st = String(row.statusKehadiran || '').trim();
                        return st !== 'Libur' && st !== 'Sakit' && st !== 'Ijin';
                    }).length;

                    const targetPJP = 80;
                    const percentage = Math.min(100, Math.round((validPJPCount / targetPJP) * 100));
                    
                    document.getElementById('pjp-progress-count').innerText = `${validPJPCount} / ${targetPJP}`;
                    document.getElementById('pjp-circular-text').innerText = `${percentage}%`;
                    document.getElementById('pjp-circular-path').setAttribute('stroke-dasharray', `${percentage}, 100`);

                    const myAssignedStores = masterStores.filter(store => String(store.tlName).trim().toLowerCase() === String(currentUser).trim().toLowerCase());
                    
                    let visitedList = [];
                    let unvisitedList = [];

                    myAssignedStores.forEach(store => {
                        const sName = String(store.namaToko || '').trim();
                        const visits = myPJPCachedRecords.filter(r => String(r.store || '').trim().toLowerCase() === sName.toLowerCase() && String(r.statusKehadiran || '').trim() === 'Check In');
                        
                        if (visits.length > 0) {
                            visitedList.push({ name: sName, code: store.kodeToko || '-', count: visits.length });
                        } else {
                            unvisitedList.push({ name: sName, code: store.kodeToko || '-', count: 0 });
                        }
                    });

                    document.getElementById('stat-visited-count').innerText = visitedList.length;
                    document.getElementById('stat-unvisited-count').innerText = unvisitedList.length;

                    let html = `<table class="w-full text-left border-collapse">
                        <thead>
                            <tr class="bg-slate-50 text-slate-600 border-b border-slate-200 text-[10px] uppercase tracking-wider">
                                <th class="p-2.5 font-bold">Waktu</th>
                                <th class="p-2.5 font-bold">Toko</th>
                                <th class="p-2.5 font-bold">Status</th>
                                <th class="p-2.5 font-bold">Catatan</th>
                            </tr>
                        </thead>
                        <tbody class="divide-y divide-slate-100">`;
                    myPJPCachedRecords.forEach(row => {
                        html += `<tr class="hover:bg-slate-50 transition">
                            <td class="p-2.5 font-medium text-slate-600">${row.date || ''} ${row.time || ''}</td>
                            <td class="p-2.5 font-medium text-slate-800">${row.store || '-'}</td>
                            <td class="p-2.5"><span class="px-2 py-0.5 bg-blue-50 text-blue-700 rounded-full text-[10px] font-bold border border-blue-100">${row.statusKehadiran || 'Check In'}</span></td>
                            <td class="p-2.5 text-slate-600">${row.notes || '-'}</td>
                        </tr>`;
                    });
                    html += `</tbody></table>`;
                    container.innerHTML = myPJPCachedRecords.length > 0 ? html : '<p class="text-slate-500 text-center py-6 font-medium">Belum ada riwayat PJP untuk Anda.</p>';
                } else {
                    document.getElementById('pjp-progress-count').innerText = `0 / 80`;
                    document.getElementById('pjp-circular-text').innerText = `0%`;
                    document.getElementById('pjp-circular-path').setAttribute('stroke-dasharray', `0, 100`);
                    document.getElementById('stat-visited-count').innerText = 0;
                    document.getElementById('stat-unvisited-count').innerText = masterStores.filter(store => String(store.tlName).trim().toLowerCase() === String(currentUser).trim().toLowerCase()).length;
                    container.innerHTML = '<p class="text-slate-500 text-center py-6 font-medium">Belum ada data PJP tercatat.</p>';
                }
            } catch(e) {
                container.innerHTML = '<p class="text-red-500 text-center py-6 font-medium">Gagal memuat riwayat.</p>';
            }
        }

        function getMyAreaPromotors() {
            let promotorSet = new Set();
            masterStores.forEach(s => {

        function populatePromotorDropdown() {
            const dropdown = document.getElementById('pantau-promotor-dropdown-list');
            const query = document.getElementById('pantau-filter-promotor').value.toLowerCase();
            
            const areaPromotors = getMyAreaPromotors();
            let promotors = areaPromotors.filter(name => name.toLowerCase().includes(query));

            dropdown.innerHTML = '';
            
            let allDiv = document.createElement('div');
            allDiv.className = "p-2.5 hover:bg-blue-50 cursor-pointer font-bold text-blue-600 text-xs";
            allDiv.innerHTML = `<i class="fa-solid fa-users mr-2"></i> Semua Promotor Area Anda`;
            allDiv.onclick = function() {
                document.getElementById('pantau-filter-promotor').value = '';
                document.getElementById('pantau-filter-promotor-selected').value = '';
                dropdown.classList.add('hidden');
                renderPantauFilteredData();
            };
            dropdown.appendChild(allDiv);

            if(promotors.length === 0) {
                let noneDiv = document.createElement('div');
                noneDiv.className = "p-2.5 text-slate-400 text-xs text-center";
                noneDiv.innerText = "Tidak ada promotor di area Anda";
                dropdown.appendChild(noneDiv);
                return;
            }

            promotors.forEach(name => {
                let div = document.createElement('div');
                div.className = "p-2.5 hover:bg-blue-50 cursor-pointer text-slate-800 text-xs font-medium";
                div.innerText = name;
                div.onclick = function() {
                    document.getElementById('pantau-filter-promotor').value = name;
                    document.getElementById('pantau-filter-promotor-selected').value = name;
                    dropdown.classList.add('hidden');
                    renderPantauFilteredData();
                };
                dropdown.appendChild(div);
            });
        }

        function renderPantauFilteredData() {
            const filterDateVal = document.getElementById('pantau-filter-date').value;
            const filterPromotorVal = document.getElementById('pantau-filter-promotor').value.trim().toLowerCase();
            
            const areaPromotors = getMyAreaPromotors().map(p => p.toLowerCase());
            const myStoreNames = masterStores
                .filter(s => String(s.tlName || '').trim().toLowerCase() === String(currentUser || '').trim().toLowerCase())
                .map(s => String(s.namaToko || '').trim().toLowerCase());

            let filteredAbsen = cachedAbsenRecords.filter(row => {
                const pName = String(row.promotor || '').trim().toLowerCase();
                const sName = String(row.store || '').trim().toLowerCase();
                return areaPromotors.includes(pName) || myStoreNames.includes(sName);
            });

            if (filterDateVal) {
                const parts = filterDateVal.split('-');
                const formattedFilter = `${parts[2]}-${parts[1]}-${parts[0]}`;
                filteredAbsen = filteredAbsen.filter(row => String(row.timestamp || '').includes(formattedFilter));
            }
            if (filterPromotorVal) {
                filteredAbsen = filteredAbsen.filter(row => String(row.promotor || '').trim().toLowerCase().includes(filterPromotorVal));
            }

            const absenContainer = document.getElementById('pantau-absen-content');
            if (filteredAbsen.length === 0) {
                absenContainer.innerHTML = '<p class="text-slate-500 text-center py-4 font-medium">Tidak ada data absensi sesuai filter.</p>';
            } else {
                let html = `<div class="space-y-2">`;
                filteredAbsen.forEach(row => {
                    html += `<div onclick='openAbsenDetail(${JSON.stringify(row)})' class="p-3 bg-slate-50 hover:bg-blue-50 border border-slate-200 rounded-2xl cursor-pointer transition flex justify-between items-center shadow-sm">
                        <div>
                            <span class="font-bold text-slate-900 block">${row.promotor || 'Promotor'} - <span class="text-blue-600">${row.status || 'Hadir'}</span></span>
                            <span class="text-[10px] text-slate-500 font-medium">${row.timestamp || ''} | Toko: ${row.store || '-'}</span>
                        </div>
                        <span class="px-2.5 py-1 bg-blue-100 text-blue-700 rounded-xl font-bold text-[10px]">Detail</span>
                    </div>`;
                });
                html += `</div>`;
                absenContainer.innerHTML = html;
            }

            let filteredSales = cachedSalesRecords.filter(row => {
                const pName = String(row.promotor || '').trim().toLowerCase();
                const sName = String(row.store || '').trim().toLowerCase();
                return areaPromotors.includes(pName) || myStoreNames.includes(sName);
            });

            if (filterDateVal) {
                const parts = filterDateVal.split('-');
                const formattedFilter = `${parts[2]}-${parts[1]}-${parts[0]}`;
                filteredSales = filteredSales.filter(row => String(row.date || '').includes(formattedFilter));
            }
            if (filterPromotorVal) {
                filteredSales = filteredSales.filter(row => String(row.promotor || '').trim().toLowerCase().includes(filterPromotorVal));
            }

            const salesContainer = document.getElementById('pantau-sales-content');
            if (filteredSales.length === 0) {
                salesContainer.innerHTML = '<p class="text-slate-500 text-center py-4 font-medium">Tidak ada data sales sesuai filter.</p>';
            } else {
                let html = `<div class="space-y-2">`;
                filteredSales.forEach(row => {
                    html += `<div onclick='openSalesDetail(${JSON.stringify(row)})' class="p-3 bg-slate-50 hover:bg-blue-50 border border-slate-200 rounded-2xl cursor-pointer transition flex justify-between items-center shadow-sm">
                        <div>
                            <span class="font-bold text-slate-900 block">${row.customer || 'Customer'} - Inv: ${row.invoice || '-'}</span>
                            <span class="text-[10px] text-slate-500 font-medium">${row.date || ''} | Toko: ${row.store || '-'} | Promotor: ${row.promotor || '-'}</span>
                        </div>
                        <span class="px-2.5 py-1 bg-blue-100 text-blue-700 rounded-xl font-bold text-[10px]">Detail</span>
                    </div>`;
                });
                html += `</div>`;
                salesContainer.innerHTML = html;
            }
        }

        async function loadPerformanceData() {
            const container = document.getElementById('performance-content');
            container.innerHTML = '<p class="text-slate-500 text-center py-6 font-medium">Memuat data performance...</p>';
            try {
                let res = await fetch(SCRIPT_URL + "?action=getPerformanceData");
                let textRes = await res.text();
                let json = JSON.parse(textRes);
                
                if(json.status === "success") {
                    const allPerformanceRows = json.data || [];
                    
                    const myAssignedStores = masterStores.filter(s => String(s.tlName || '').trim().toLowerCase() === String(currentUser || '').trim().toLowerCase());
                    const myAssignedStoreNames = myAssignedStores.map(s => String(s.namaToko || '').trim().toLowerCase());

                    cachedPerformanceRows = allPerformanceRows.filter(r => {
                        const sName = String(r.storeName || '').trim().toLowerCase();
                        return myAssignedStoreNames.some(assigned => sName.includes(assigned) || assigned.includes(sName));
                    });
                    
                    let totalGA = 0;
                    let totalCPP = 0;
                    let totalRevenue = 0;
                    let totalMtdGA = 0;
                    let totalMtdCPP = 0;
                    let totalMtdTotal = 0;
                    let totalMtdRevenue = 0;
                    let storeZeroGACount = 0;

                    cachedPerformanceRows.forEach(r => {
                        totalGA += Number(r.targetGA || 0);
                        totalCPP += Number(r.targetCPP || 0);
                        totalRevenue += Number(r.targetRevenue || 0);
                        totalMtdGA += Number(r.mtdGA || 0);
                        totalMtdCPP += Number(r.mtdCPP || 0);
                        totalMtdTotal += Number(r.mtdTotal || 0);
                        totalMtdRevenue += Number(r.mtdRevenue || 0);
                        if (Number(r.mtdGA || 0) === 0) {
                            storeZeroGACount++;
                        }
                    });

                    const totalTarget = totalGA + totalCPP;
                    const milestoneVal = totalTarget > 0 ? Math.round((totalMtdTotal / totalTarget) * 100) : 0;

                    document.getElementById('target-ga-val').innerText = totalGA.toLocaleString('id-ID');
                    document.getElementById('target-cpp-val').innerText = totalCPP.toLocaleString('id-ID');
                    document.getElementById('target-total-val').innerText = totalTarget.toLocaleString('id-ID');
                    document.getElementById('target-rev-val').innerText = 'Rp ' + totalRevenue.toLocaleString('id-ID');

                    document.getElementById('mtd-ga-val').innerText = totalMtdGA.toLocaleString('id-ID');
                    document.getElementById('mtd-cpp-val').innerText = totalMtdCPP.toLocaleString('id-ID');

        function populatePerfPromotorDropdown() {
            const dropdown = document.getElementById('perf-promotor-dropdown-list');
            const query = document.getElementById('perf-promotor-search').value.toLowerCase();
            
            const areaPromotors = getMyAreaPromotors();
            let promotors = areaPromotors.filter(name => name.toLowerCase().includes(query));

            dropdown.innerHTML = '';
            
            let allDiv = document.createElement('div');
            allDiv.className = "p-2.5 hover:bg-blue-50 cursor-pointer font-bold text-blue-600 text-xs";
            allDiv.innerHTML = `<i class="fa-solid fa-users mr-2"></i> Semua Promotor Area Anda`;
            allDiv.onclick = function() {
                document.getElementById('perf-promotor-search').value = '';
                document.getElementById('perf-promotor-selected-val').value = '';
                dropdown.classList.add('hidden');
                renderPromotorPerformanceView();
            };
            dropdown.appendChild(allDiv);

            if(promotors.length === 0) {
                let noneDiv = document.createElement('div');
                noneDiv.className = "p-2.5 text-slate-400 text-xs text-center";
                noneDiv.innerText = "Tidak ada promotor ditemukan";
                dropdown.appendChild(noneDiv);
                return;
            }

            promotors.forEach(name => {
                let div = document.createElement('div');
                div.className = "p-2.5 hover:bg-blue-50 cursor-pointer text-slate-800 text-xs font-medium";

                div.onclick = function() {
                    document.getElementById('perf-promotor-search').value = name;
                    document.getElementById('perf-promotor-selected-val').value = name;
                    dropdown.classList.add('hidden');
                    renderPromotorPerformanceView();
                };
                dropdown.appendChild(div);
            });
        }

        function getRowsForSelectedPromotor() {
            const selectedPromotor = document.getElementById('perf-promotor-search').value.trim().toLowerCase();
            let matchingStores = masterStores.filter(s => String(s.tlName || '').trim().toLowerCase() === String(currentUser || '').trim().toLowerCase());
            if (selectedPromotor) {
                matchingStores = matchingStores.filter(s => String(s.promotor || '').trim().toLowerCase().includes(selectedPromotor));
            }
            const matchingStoreNames = matchingStores.map(s => String(s.namaToko || '').trim().toLowerCase());

            let rowsToSum = cachedPerformanceRows;
            if (selectedPromotor) {
                rowsToSum = cachedPerformanceRows.filter(r => {
                    const rName = String(r.storeName || '').trim().toLowerCase();
                    return matchingStoreNames.some(m => rName.includes(m) || m.includes(rName));
                });
            }
            return rowsToSum;
        }

        function renderPromotorPerformanceView() {
            const rowsToSum = getRowsForSelectedPromotor();

            let sumGA = 0, sumCPP = 0, sumRevenue = 0, sumMtdGA = 0, sumMtdCPP = 0, sumMtdTotal = 0, sumMtdRevenue = 0;
            rowsToSum.forEach(r => {
                sumGA += Number(r.targetGA || 0);
                sumCPP += Number(r.targetCPP || 0);
                sumRevenue += Number(r.targetRevenue || 0);
                sumMtdGA += Number(r.mtdGA || 0);
                sumMtdCPP += Number(r.mtdCPP || 0);
                sumMtdTotal += Number(r.mtdTotal || 0);
                sumMtdRevenue += Number(r.mtdRevenue || 0);
            });

            const sumTotalTarget = sumGA + sumCPP;
            const sumMilestone = sumTotalTarget > 0 ? Math.round((sumMtdTotal / sumTotalTarget) * 100) : 0;

            document.getElementById('p-card-ga').innerText = sumGA.toLocaleString('id-ID');
            document.getElementById('p-card-cpp').innerText = sumCPP.toLocaleString('id-ID');
            document.getElementById('p-card-total').innerText = sumTotalTarget.toLocaleString('id-ID');
            document.getElementById('p-card-rev').innerText = 'Rp ' + sumRevenue.toLocaleString('id-ID');
            document.getElementById('p-card-mtd-ga').innerText = sumMtdGA.toLocaleString('id-ID');
            document.getElementById('p-card-mtd-cpp').innerText = sumMtdCPP.toLocaleString('id-ID');
            document.getElementById('p-card-mtd-total').innerText = sumMtdTotal.toLocaleString('id-ID');
            document.getElementById('p-card-milestone').innerText = sumMilestone + '%';
        }

        function renderPromotorLeaderboard() {
            const container = document.getElementById('promotor-leaderboard-container');
            const areaPromotors = getMyAreaPromotors();

            if (areaPromotors.length === 0) {
                container.innerHTML = `<p class="text-[11px] text-slate-400 text-center py-2 font-medium">Tidak ada promotor tercatat di area Anda.</p>`;
                return;
            }

            let promotorStats = areaPromotors.map(pName => {
                const pStores = masterStores.filter(s => String(s.tlName || '').trim().toLowerCase() === String(currentUser || '').trim().toLowerCase() && String(s.promotor || '').trim().toLowerCase() === pName.toLowerCase());
                const pStoreNames = pStores.map(s => String(s.namaToko || '').trim().toLowerCase());

                const pRows = cachedPerformanceRows.filter(r => {
                    const rName = String(r.storeName || '').trim().toLowerCase();
                    return pStoreNames.some(m => rName.includes(m) || m.includes(rName));
                });

                let mtdGA = 0, mtdCPP = 0, mtdTotal = 0, targetGA = 0, targetCPP = 0;
                pRows.forEach(r => {
                    mtdGA += Number(r.mtdGA || 0);
                    mtdCPP += Number(r.mtdCPP || 0);
                    mtdTotal += Number(r.mtdTotal || 0);
                    targetGA += Number(r.targetGA || 0);
                    targetCPP += Number(r.targetCPP || 0);
                });

                const totTarget = targetGA + targetCPP;
                const milestone = totTarget > 0 ? Math.round((mtdTotal / totTarget) * 100) : 0;

                return { name: pName, mtdGA, mtdCPP, mtdTotal, milestone };
            });

            promotorStats.sort((a, b) => b.milestone - a.milestone);

            let html = `<table class="w-full text-left border-collapse bg-white rounded-2xl overflow-hidden border border-slate-200 shadow-sm text-xs">
                <thead>
                    <tr class="bg-slate-100 text-slate-700 border-b border-slate-200 text-[10px] uppercase tracking-wider">
                        <th class="p-2.5 font-bold">Promotor</th>
                        <th class="p-2.5 font-bold text-center">MTD GA</th>
                        <th class="p-2.5 font-bold text-center">MTD CPP</th>
                        <th class="p-2.5 font-bold text-center">Total</th>
                        <th class="p-2.5 font-bold text-right">Milestone</th>
                    </tr>
                </thead>
                <tbody class="divide-y divide-slate-100">`;

            promotorStats.forEach((p, idx) => {
                let rankBadge = '';
                if (idx === 0) rankBadge = `<span class="w-5 h-5 rounded-full bg-amber-400 text-white font-bold inline-flex items-center justify-center text-[9px] mr-1 shadow-sm"><i class="fa-solid fa-crown text-[8px]"></i></span>`;
                else if (idx === 1) rankBadge = `<span class="w-5 h-5 rounded-full bg-slate-300 text-slate-800 font-bold inline-flex items-center justify-center text-[9px] mr-1">2</span>`;
                else if (idx === 2) rankBadge = `<span class="w-5 h-5 rounded-full bg-amber-700/30 text-amber-800 font-bold inline-flex items-center justify-center text-[9px] mr-1">3</span>`;
                else rankBadge = `<span class="w-5 h-5 rounded-full bg-slate-100 text-slate-600 font-bold inline-flex items-center justify-center text-[9px] mr-1">${idx + 1}</span>`;

                html += `<tr class="hover:bg-slate-50 transition">
                    <td class="p-2.5 font-bold text-slate-900 flex items-center">${rankBadge} ${p.name}</td>
                    <td class="p-2.5 text-center font-semibold text-blue-600">${p.mtdGA.toLocaleString('id-ID')}</td>
                    <td class="p-2.5 text-center font-semibold text-indigo-600">${p.mtdCPP.toLocaleString('id-ID')}</td>
                    <td class="p-2.5 text-center font-semibold text-emerald-600">${p.mtdTotal.toLocaleString('id-ID')}</td>
                    <td class="p-2.5 text-right font-black text-blue-700">${p.milestone}%</td>
                </tr>`;
            });

            html += `</tbody></table>`;
            container.innerHTML = html;
        }

        function filterMetricModalList() {
            const query = document.getElementById('metric-search-input').value.trim().toLowerCase();
            let baseRows = cachedPerformanceRows;
            if (currentActiveMetricType.startsWith('promotor_')) {
                baseRows = getRowsForSelectedPromotor();
            } else if (currentActiveMetricType === 'store_zero_ga') {
                baseRows = cachedPerformanceRows.filter(r => Number(r.mtdGA || 0) === 0);
            }
            const filtered = baseRows.filter(r => String(r.storeName || '').toLowerCase().includes(query));
            renderMetricModalList(filtered);
        }

        function renderMetricModalList(rows) {
            const bodyEl = document.getElementById('metric-modal-body');
            if (rows.length === 0) {
                bodyEl.innerHTML = `<p class="text-slate-500 text-center py-4 font-medium">Tidak ada data toko ditemukan.</p>`;
                return;
            }

            let html = '';
            rows.forEach(r => {
                const ga = Number(r.targetGA || 0);
                const cpp = Number(r.targetCPP || 0);
                const total = ga + cpp;
                const rev = Number(r.targetRevenue || 0);

                const mtdGa = Number(r.mtdGA || 0);
                const mtdCpp = Number(r.mtdCPP || 0);
                const mtdTot = Number(r.mtdTotal || 0);
                const mtdRev = Number(r.mtdRevenue || 0);
                const storeMilestone = total > 0 ? Math.round((mtdTot / total) * 100) : 0;

                const baseMetric = currentActiveMetricType.replace('promotor_', '');

                let valDisplay = '';
                if (baseMetric === 'ga') valDisplay = `<span class="font-black text-blue-600 text-sm">${ga.toLocaleString('id-ID')}</span>`;
                else if (baseMetric === 'cpp') valDisplay = `<span class="font-black text-indigo-600 text-sm">${cpp.toLocaleString('id-ID')}</span>`;
                else if (baseMetric === 'total') valDisplay = `<span class="font-black text-emerald-600 text-sm">${total.toLocaleString('id-ID')}</span>`;
                else if (baseMetric === 'revenue') valDisplay = `<span class="font-black text-amber-600 text-sm">Rp ${rev.toLocaleString('id-ID')}</span>`;
                else if (baseMetric === 'mtd_ga') valDisplay = `<span class="font-black text-slate-800 text-sm">${mtdGa.toLocaleString('id-ID')} <span class="text-[9px] text-slate-400 font-normal">/ target ${ga}</span></span>`;
                else if (baseMetric === 'mtd_cpp') valDisplay = `<span class="font-black text-slate-800 text-sm">${mtdCpp.toLocaleString('id-ID')} <span class="text-[9px] text-slate-400 font-normal">/ target ${cpp}</span></span>`;
                else if (baseMetric === 'mtd_total') valDisplay = `<span class="font-black text-slate-800 text-sm">${mtdTot.toLocaleString('id-ID')} <span class="text-[9px] text-slate-400 font-normal">/ target ${total}</span></span>`;
                else if (baseMetric === 'mtd_revenue') valDisplay = `<span class="font-black text-slate-800 text-sm">Rp ${mtdRev.toLocaleString('id-ID')} <span class="text-[9px] text-slate-400 font-normal">/ target Rp ${rev.toLocaleString('id-ID')}</span></span>`;
                else if (baseMetric === 'milestone') valDisplay = `<span class="font-black text-blue-700 text-sm">${storeMilestone}% <span class="text-[9px] text-slate-400 font-normal">(${mtdTot}/${total})</span></span>`;
                else if (baseMetric === 'store_zero_ga') valDisplay = `<span class="font-black text-rose-600 text-sm">MTD GA = ${mtdGa}</span>`;

                html += `<div class="py-2.5 flex justify-between items-center">
                    <div>
                        <span class="font-bold text-slate-900 block">${r.storeName || '-'}</span>
                        <span class="text-[10px] text-slate-400">GA: ${ga} | CPP: ${cpp} | Rev: Rp ${rev.toLocaleString('id-ID')}</span>
                    </div>
                    <div>${valDisplay}</div>
                </div>`;
            });
            bodyEl.innerHTML = html;
        }

        async function submitSales(event) {
            event.preventDefault();
            const storeName = document.getElementById('sales-store-select').value;
            const matchedStore = masterStores.find(s => s.namaToko === storeName);

            const payload = {
                type: "sales",
                data: {
                    uniqueCode: document.getElementById('sales-unique-code').value,
                    customer: document.getElementById('sales-customer').value,
                    date: document.getElementById('sales-date').value,
                    store: storeName,
                    kodeToko: matchedStore ? matchedStore.kodeToko : '',
                    channel: matchedStore ? matchedStore.channel : '',
                    invoice: document.getElementById('sales-invoice').value,
                    paket: document.getElementById('sales-paket').value,
                    kontrak: document.getElementById('sales-kontrak').value,
                    kategori: document.getElementById('sales-kategori').value,
                    status: document.getElementById('sales-status').value,
                    remarks: document.getElementById('sales-remarks').value,
                    promotor: currentUser
                }
            };

            toggleLoading(true, "Mengirim Report Sales...");
            try {
                await fetch(SCRIPT_URL, {
                    method: "POST",
                    mode: "no-cors",
                    headers: { "Content-Type": "text/plain" },
                    body: JSON.stringify(payload)
                });
                toggleLoading(false);
                showModal('Berhasil', 'Daily Report Sales berhasil dikirim!');
                document.getElementById('sales-form').reset();
                document.getElementById('sales-date').value = getFormattedDate(new Date());
                generateUniqueCodes();
            } catch(err) {
                toggleLoading(false);
                showModal('Gagal', 'Terjadi kesalahan koneksi.');
            }
        }

            let visibilityPOP = "-";

                    showModal('Perhatian', 'Silakan pilih toko kunjungan terlebih dahulu.');

                    time: document.getElementById('pjp-time').value,
                    tlName: document.getElementById('pjp-tl-name').value,

                    store: storeName,
                    kodeToko: kodeToko,
                    region: document.getElementById('pjp-region').value,
                    promotor: promotorName,
                    stokKartu: stokKartu,
                    visibilitySP: visibilitySP,
                    visibilityPOP: visibilityPOP,
                    grooming: grooming,
                    notes: document.getElementById('pjp-notes').value,
